# engine_platform: native windows (plan F2 Task 6). Win32 only for now.
if(WIN32)
    add_library(engine_platform STATIC Window.h win32/Win32Window.cpp)
    add_library(engine::platform ALIAS engine_platform)
    target_include_directories(engine_platform PUBLIC "${PROJECT_SOURCE_DIR}/engine")
    target_link_libraries(engine_platform PUBLIC engine::rhi PRIVATE user32 shell32)
    target_compile_definitions(engine_platform PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    if(MSVC)
        target_compile_options(engine_platform PRIVATE /W4 /WX /permissive- /utf-8 /external:W0)
    endif()
endif()
